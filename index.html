// ========== 前処理: 画像拡大のみ（過度な二値化は削除） ========== function preprocessImage(file) { return new Promise((resolve) => { const img = new Image(); img.onload = () => { const canvas = document.createElement('canvas'); const ctx = canvas.getContext('2d'); // 解像度を上げて読み取りやすくする（拡大のみ行う） const scale = 2; canvas.width = img.width * scale; canvas.height = img.height * scale; ctx.drawImage(img, 0, 0, canvas.width, canvas.height); resolve(canvas.toDataURL('image/png')); }; img.src = URL.createObjectURL(file); }); } // ========== レーベンシュタイン距離を用いた曖昧マッチング（部分文字列スライド判定） ========== function matchSkillsFuzzy(ocrText) { // 改行・スペース・記号を全て取り除いた1つの連続文字列にする const normalizedOcrText = ocrText.replace(/[\s★☆|:;.\-\n]/g, ''); skillList.forEach(skill => { const targetSkill = skill.replace(/\s+/g, ''); if (!targetSkill) return; // 1. 完全部分一致チェック if (normalizedOcrText.includes(targetSkill)) { detectedSkills.add(skill); return; } // 2. 部分文字列のスライド比較（N-gram的にスキル名長＋前後1文字で切り出して比較） const skillLen = targetSkill.length; if (skillLen === 0) return; for (let i = 0; i <= normalizedOcrText.length - skillLen; i++) { // スキル名とほぼ同じ長さの区間を切り出す const chunk = normalizedOcrText.substr(i, skillLen); const similarity = getSimilarity(chunk, targetSkill); // 70%以上の類似度があれば検出判定 if (similarity >= 0.70) { detectedSkills.add(skill); break; } } }); }
